\tikzset{femm/materials/test tensor/.style={mu x=2+1,mu y=7,material angle=23}}
\tikzpicture[femm/problem={xmin=0,xmax=20,ymin=0,ymax=12,mesh size=3,depth=5},
  femm/pair={first=a,second=b,shift x=20},
  femm/boundary={side=top,value=.002},femm/cache={file=static-cache,mode=\CacheMode},scale=.8]
  \path[femm/domain] (0,0) rectangle (20,12);
  % Explicit path tags replace the domain's default A=0 on the top edge.
  \path[femm/frontier={name=top,value=.002}] (0,12)--(20,12);
  \path[femm/frontier={name=a,type=periodic}] (0,0)--(0,4)--(0,12);
  \path[femm/frontier={name=b,type=periodic}] (20,0)--(20,8)--(20,12);
  \path[femm/region={material=test tensor}] (0,0) rectangle (20,12);
  \pic[femm/lines=5,femm/arrows=false,femm/repeat={copies=2}] {femm repeated field};
  \directlua{
    assert(luafemm.current.cache_info.status=='\ExpectedCache')
    local bx,by=luafemm.sample(luafemm.current,10,6)
    assert(math.abs(bx-1/6)<1e-6 and math.abs(by)<1e-6)
    texio.write_nl('LUAFEMM-STATIC: '..bx)
  }
\endtikzpicture
