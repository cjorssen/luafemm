% Identical geometry and assertions for all three format wrappers.
\filldraw[femm/region={material=cast-alnico-5-lng37,magnetization angle=90},
  fill=red!20] (-3,-2) rectangle (3,2);
\filldraw[femm/region={material=pure-iron},fill=gray!20]
  (-3,-4) rectangle (3,-3);
\path[femm/region={material=copper}] (5,-2) rectangle (6,2);
\draw[femm/profile={name=format-test,samples=17,curve tolerance=.01},blue]
  (-5,4) .. controls (-2,7) and (2,7) .. (5,4);
\pic[femm/lines=9] {femm field};
\directlua{
  assert(luafemm.version == '\luafemmversion')
  assert(luafemm.current.stats.residual < 1e-7)
  local p = require('luafemm-profiles').sample('format-test')
  assert(rawlen(p) == 17)
  assert(p[9].By > 0)
  texio.write_nl('LUAFEMM-SAMPLE-BX: '..p[9].Bx)
  texio.write_nl('LUAFEMM-SAMPLE-BY: '..p[9].By)
  texio.write_nl('LUAFEMM-SAMPLE-NODES: '..luafemm.current.stats.nodes)
}
