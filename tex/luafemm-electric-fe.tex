% One dispatcher transports only declared string values, never arbitrary Lua.
\def\luafemm@electricphysical#1{\begingroup
  \def\luafemm@physicalvalue##1{\pgfkeysvalueof{/femm/electric #1/##1}}%
  \pgfsyssoftpath@getcurrentpath\luafemm@electricpath
  \pgfprocessround{\luafemm@electricpath}{\luafemm@electricpath}%
  \ifpgfsys@eorule\def\luafemm@electricfill{even odd}\else\def\luafemm@electricfill{nonzero}\fi
  \def\luafemm@physicalextra{}%
  \def\luafemm@physicalkind{#1}\def\luafemm@physicaltest{region}%
  \ifx\luafemm@physicalkind\luafemm@physicaltest
    \def\luafemm@physicalextra{material='\luaescapestring{\luafemm@physicalvalue{material}}',epsilon_r='\luaescapestring{\luafemm@physicalvalue{relative permittivity}}',
      epsilon_x='\luaescapestring{\luafemm@physicalvalue{relative permittivity x}}',
      epsilon_y='\luaescapestring{\luafemm@physicalvalue{relative permittivity y}}',
      material_angle='\luaescapestring{\luafemm@physicalvalue{material angle}}',
      density='\luaescapestring{\luafemm@physicalvalue{volume charge density}}',}%
  \fi
  \def\luafemm@physicaltest{electrode}%
  \ifx\luafemm@physicalkind\luafemm@physicaltest
    \def\luafemm@physicalextra{name='\luaescapestring{\luafemm@physicalvalue{name}}',
      potential='\luaescapestring{\luafemm@physicalvalue{potential}}',
      interior='\luaescapestring{\luafemm@physicalvalue{interior}}',
      condition='\luaescapestring{\luafemm@physicalvalue{condition}}',
      charge='\luaescapestring{\luafemm@physicalvalue{charge}}',}%
  \fi
  \def\luafemm@physicaltest{charge layer}%
  \ifx\luafemm@physicalkind\luafemm@physicaltest
    \def\luafemm@physicalextra{density='\luaescapestring{\luafemm@physicalvalue{surface charge density}}',}%
  \fi
  \def\luafemm@physicaltest{boundary}%
  \ifx\luafemm@physicalkind\luafemm@physicaltest
    \def\luafemm@physicalextra{condition='\luaescapestring{\luafemm@physicalvalue{condition}}',
      potential='\luaescapestring{\luafemm@physicalvalue{potential}}',
      displacement='\luaescapestring{\luafemm@physicalvalue{normal displacement}}',
      coefficient='\luaescapestring{\luafemm@physicalvalue{coefficient}}',
      pair='\luaescapestring{\luafemm@physicalvalue{pair}}',}%
  \fi
  \directlua{require('luafemm-electric-tex').physical('\luafemm@electricid','#1',
    '\luaescapestring{\meaning\luafemm@electricpath}',{\luafemm@physicalextra
      mesh_size='\luaescapestring{\luafemm@physicalvalue{mesh size}}',
      tolerance='\luaescapestring{\luafemm@physicalvalue{curve tolerance}}',
      fill_rule='\luafemm@electricfill'})}%
  \endgroup}
