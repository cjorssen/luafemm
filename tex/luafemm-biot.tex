\def\luafemm@biotstart{%
 \ifx\luafemm@biotid\pgfutil@empty
  \ifluafemm@modelstarted\pgferror{luafemm: use a separate Biot-Savart picture}\fi
  \ifx\luafemm@electricid\pgfutil@empty\else\pgferror{luafemm: use a separate Biot-Savart picture}\fi
  \pgfgettransform\luafemm@biottransform
  \directlua{require('luafemm-biot-tex').new({
  length_unit='\luaescapestring{\pgfkeysvalueof{/femm/biot/length unit}}',
  mu_r='\luaescapestring{\pgfkeysvalueof{/femm/biot/relative permeability}}',
  tolerance='\luaescapestring{\pgfkeysvalueof{/femm/biot/quadrature tolerance}}',
  absolute_tolerance='\luaescapestring{\pgfkeysvalueof{/femm/biot/absolute tolerance}}',
  max_depth='\luaescapestring{\pgfkeysvalueof{/femm/biot/maximum quadrature depth}}',
  max_evaluations='\luaescapestring{\pgfkeysvalueof{/femm/biot/maximum evaluations}}',
  exclusion='\luaescapestring{\pgfkeysvalueof{/femm/biot/exclusion radius}}',
  native_file='\luaescapestring{\pgfkeysvalueof{/femm/biot/native file}}',
  name='\luaescapestring{\pgfkeysvalueof{/femm/biot/name}}',explicit='\luafemm@biotexplicit'},
 '\luaescapestring{\luafemm@biottransform}',
 {'\the\pgf@xx','\the\pgf@xy','\the\pgf@yx','\the\pgf@yy'})}%
 \else\pgferror{luafemm: nested Biot-Savart pictures are unsupported}\fi}
