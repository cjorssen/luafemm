  \edef\luafemm@electricnumber{\directlua{require('luafemm-electric-tex').value(
    '\luafemm@electricid',
    '\luaescapestring{\pgfkeysvalueof{/femm/electric value/model}}',
    '\luaescapestring{\pgfkeysvalueof{/femm/electric value/quantity}}',
    '\luaescapestring{\pgfkeysvalueof{/femm/electric value/electrode}}')}}%
  \pgfkeysgetvalue{/femm/electric value/number format}\luafemm@electricformat
  \expandafter\pgfmathprintnumber\expandafter[\luafemm@electricformat]{\luafemm@electricnumber}%
  \endgroup}
\endinput
