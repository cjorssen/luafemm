\pgfkeys{/femm/selection/.is family,/femm/selection,
  name/.initial=,material/.initial=,curve tolerance/.initial=.03}
\def\luafemm@selectionvalue#1{\pgfkeysvalueof{/femm/selection/#1}}
\tikzset{femm/selection/.code={%
  \pgfqkeys{/femm/selection}{#1}%
  \tikzset{postaction={femm/capture selection}}},
  femm/capture selection/.code={\tikz@addmode{\luafemm@captureselection}},
  femm/select/.code={\femmselect[#1]},
  femm/measure/.code={\femmmeasure[#1]}}
\def\luafemm@captureselection{%
  \begingroup
  \pgfsyssoftpath@getcurrentpath\luafemm@selectionpath
  \pgfprocessround{\luafemm@selectionpath}{\luafemm@selectionpath}%
  \def\luafemm@selectionrule{nonzero}%
  \ifpgfsys@eorule\def\luafemm@selectionrule{even odd}\fi
  \directlua{require('luafemm-post').capture(luafemm.current,
    '\luaescapestring{\luafemm@selectionvalue{name}}',
    '\luaescapestring{\meaning\luafemm@selectionpath}',{
    material='\luaescapestring{\luafemm@selectionvalue{material}}',
    tolerance=luafemm.tex.number('\luaescapestring{\luafemm@selectionvalue{curve tolerance}}','curve tolerance'),
    fill_rule='\luafemm@selectionrule'})}%
  \endgroup}
\protected\def\femmselect{\pgfutil@ifnextchar[{\luafemm@select}{\luafemm@select[]}}
\def\luafemm@select[#1]{\begingroup
  \pgfqkeys{/femm/selection}{name={},material={},#1}%
  \directlua{require('luafemm-post').select(luafemm.current,
    '\luaescapestring{\luafemm@selectionvalue{name}}',{
    material='\luaescapestring{\luafemm@selectionvalue{material}}'})}%
  \endgroup}
\pgfkeys{/femm/measure/.is family,/femm/measure,
  name/.initial=,selection/.initial=,profile/.initial=,circuit/.initial=,
  origin x/.initial=0,origin y/.initial=0,parameter/.initial=0,tolerance/.initial=1e-10,
  quantity/.is choice,
  quantity/integrals/.code={\def\luafemm@measurequantity{integrals}},
  quantity/circuit/.code={\def\luafemm@measurequantity{circuit}},
  quantity/force/.code={\def\luafemm@measurequantity{force}},
  method/.is choice,
  method/weighted/.code={\def\luafemm@measuremethod{weighted}},
  method/maxwell/.code={\def\luafemm@measuremethod{maxwell}},
  method/lorentz/.code={\def\luafemm@measuremethod{lorentz}},
  weighting/.is choice,
  weighting/midgap/.code={\def\luafemm@measureweight{midgap}},
  weighting/area/.code={\def\luafemm@measureweight{area}},
  weighting/harmonic/.code={\def\luafemm@measureweight{harmonic}}}
\def\luafemm@measurevalue#1{\pgfkeysvalueof{/femm/measure/#1}}
\protected\def\femmmeasure{\pgfutil@ifnextchar[{\luafemm@measure}{\luafemm@measure[]}}
\def\luafemm@measure[#1]{\begingroup
  \pgfqkeys{/femm/measure}{name={},selection={},profile={},circuit={},origin x=0,origin y=0,
    parameter=0,tolerance=1e-10,quantity=integrals,method=weighted,weighting=midgap,#1}%
  \directlua{require('luafemm-post').measure(luafemm.current,{
    name='\luaescapestring{\luafemm@measurevalue{name}}',
    selection='\luaescapestring{\luafemm@measurevalue{selection}}',
    profile='\luaescapestring{\luafemm@measurevalue{profile}}',
    circuit='\luaescapestring{\luafemm@measurevalue{circuit}}',
    quantity='\luafemm@measurequantity',method='\luafemm@measuremethod',weighting='\luafemm@measureweight',
    origin_x=luafemm.tex.number('\luaescapestring{\luafemm@measurevalue{origin x}}','origin x'),
    origin_y=luafemm.tex.number('\luaescapestring{\luafemm@measurevalue{origin y}}','origin y'),
    parameter=luafemm.tex.number('\luaescapestring{\luafemm@measurevalue{parameter}}','parameter'),
    tolerance=luafemm.tex.number('\luaescapestring{\luafemm@measurevalue{tolerance}}','tolerance')})}%
  \endgroup}
% Expandable, unformatted SI scalar for use in ordinary PGF calculations.
\def\femmresult#1#2{\directlua{tex.sprint(require('luafemm-post').formatted(
  '\luaescapestring{#1}','\luaescapestring{#2}'))}}
\pgfkeys{/femm/value/.is family,/femm/value,
  component/.initial=energy,number format/.code={\pgfqkeys{/pgf/number format}{#1}},unit/.initial=auto}
\protected\def\femmvalue{\pgfutil@ifnextchar[{\luafemm@value}{\luafemm@value[]}}
\def\luafemm@value[#1]#2{\begingroup
  \pgfqkeys{/femm/value}{component=energy,number format={precision=3},unit=auto,#1}%
  \edef\luafemm@valueunit{\pgfkeysvalueof{/femm/value/unit}}\def\luafemm@autounit{auto}%
  \ifx\luafemm@valueunit\luafemm@autounit
    \edef\luafemm@valueunit{\directlua{tex.sprint(require('luafemm-post').unit(
      '\luaescapestring{\pgfkeysvalueof{/femm/value/component}}'))}}%
  \fi
  \pgfmathprintnumber{\femmresult{#2}{\pgfkeysvalueof{/femm/value/component}}}%
  \ifx\luafemm@valueunit\pgfutil@empty\else\ifmmode\,{\rm\luafemm@valueunit}\else$\,{\rm\luafemm@valueunit}$\fi\fi
  \endgroup}
\pgfkeys{/femm/result plot/.is family,/femm/result plot,
  results/.initial=,component/.initial=force,abscissa/.initial=parameter,plot options/.initial={}}
\protected\def\femmresultplot{\pgfutil@ifnextchar[{\luafemm@resultplot}{\luafemm@resultplot[]}}
\def\luafemm@resultplot[#1]{%
  \ifdefined\addplot
    \begingroup
    \pgfqkeys{/femm/result plot}{results={},component=force,abscissa=parameter,plot options={},#1}%
    \pgfkeysgetvalue{/femm/result plot/plot options}\luafemm@plotoptions
    \luafemm@plottoks=\expandafter{\luafemm@plotoptions}%
    \edef\luafemm@plotcommand{\endgroup
      \noexpand\addplot+[\the\luafemm@plottoks] coordinates {\directlua{tex.sprint(
        require('luafemm-post').coordinates('\luaescapestring{\pgfkeysvalueof{/femm/result plot/results}}',
        '\luaescapestring{\pgfkeysvalueof{/femm/result plot/component}}',
        '\luaescapestring{\pgfkeysvalueof{/femm/result plot/abscissa}}'))}};}%
    \luafemm@plotcommand
  \else\errmessage{luafemm: femmresultplot requires pgfplots and an axis environment}\fi}

\def\luafemmForceLabel#1{%
  \pgfkeysgetvalue{/femm/force/number format}\luafemm@forceformat
  \expandafter\pgfmathprintnumber\expandafter[\luafemm@forceformat]{#1}$\,{\rm N}$}
\tikzset{femm/force/.code={\pgfqkeys{/femm/force}{#1}},
  femm force arrow/.style={-{Stealth[length=2mm]},draw=orange!85!black,line width=1pt},
  femm force label/.style={pos=1,anchor=south west,fill=white,fill opacity=.9,text opacity=1,
    rounded corners=1pt,inner sep=2pt,font=\pgfutil@font@scriptsize,text=black},
  pics/femm force/.style={code={\begingroup\pgftransformreset
    \directlua{local n=luafemm.tex.number
      local x='\luaescapestring{\pgfkeysvalueof{/femm/force/x}}'
      local y='\luaescapestring{\pgfkeysvalueof{/femm/force/y}}'
      require('luafemm-post').arrow(luafemm.current,
        '\luaescapestring{\pgfkeysvalueof{/femm/force/result}}',
        n('\luaescapestring{\pgfkeysvalueof{/femm/force/length}}','force length'),
        n('\luaescapestring{\pgfkeysvalueof{/femm/force/scale}}','force scale'),
        not(x=='') and n(x,'force x') or nil,not(y=='') and n(y,'force y') or nil,
        '\luaescapestring{\pgfkeysvalueof{/femm/force/label}}')}%
    \endgroup}}}
