  \begingroup\pgfqkeys{/femm/harmonic}{#1}%
  \pgfkeysgetvalue{/femm/harmonic/plot options}\luafemm@plotoptions
  \luafemm@plottoks=\expandafter{\luafemm@plotoptions}%

    \noexpand\addplot+[\the\luafemm@plottoks] coordinates {%
      \directlua{luafemm.tex.harmonic({\luafemm@harmonicoptions},true)}};}%
  \luafemm@next}
