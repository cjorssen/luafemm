\let\luafemm@pendingexcitations\pgfutil@empty
\def\luafemm@excitation#1{%
  \begingroup\pgfqkeys{/femm/excitation}{x=0,y=0,ampere turns=0,#1}%
  \directlua{luafemm.excitation(luafemm.current,
    luafemm.tex.number('\luaescapestring{\pgfkeysvalueof{/femm/excitation/x}}','excitation x'),
    luafemm.tex.number('\luaescapestring{\pgfkeysvalueof{/femm/excitation/y}}','excitation y'),
    luafemm.tex.number('\luaescapestring{\pgfkeysvalueof{/femm/excitation/ampere turns}}','ampere turns'))}%
  \endgroup}
\tikzset{
  femm/excitation/.code={%
    \ifluafemm@modelstarted
      \luafemm@excitation{#1}%
    \else
      \expandafter\def\expandafter\luafemm@pendingexcitations\expandafter{%
        \luafemm@pendingexcitations\luafemm@excitation{#1}}%
    \fi},
  femm/mean/.code={\pgfqkeys{/femm/mean}{#1}},
  femm mean path/.style={draw=red!75!black,thick,dashed,femm oriented field},
  femm current mean/.style={},
  femm/mean style/.code={\tikzset{femm current mean/.style={#1}}},
  pics/femm mean path/.style={code={%
    \begingroup\pgftransformreset
    \directlua{luafemm.tex.mean_picture(luafemm.current,{
      component='\luaescapestring{\pgfkeysvalueof{/femm/mean/component}}',
      cycle='\luaescapestring{\pgfkeysvalueof{/femm/mean/cycle}}',
      kind='\luaescapestring{\pgfkeysvalueof{/femm/mean/kind}}',
      profile='\luaescapestring{\pgfkeysvalueof{/femm/mean/profile}}',
      samples=luafemm.tex.number('\luaescapestring{\pgfkeysvalueof{/femm/mean/samples}}','mean samples')})}%
    \endgroup}}
}
% Use PGF's number printer, including its normal number-format styles.
\protected\def\femmflux{\pgfutil@ifnextchar[{\luafemm@flux}{\luafemm@flux[]}}
\def\luafemm@flux[#1]{%
  \begingroup\pgfqkeys{/femm/flux}{profile=,#1}%
  \edef\luafemm@fluxnumber{\directlua{
    luafemm.tex.flux_number('\luaescapestring{\pgfkeysvalueof{/femm/flux/profile}}',
      luafemm.tex.number('\luaescapestring{\pgfkeysvalueof{/femm/flux/snap tolerance}}','snap tolerance'))}}%
  \pgfmathprintnumber{\luafemm@fluxnumber}%
  \endgroup}
\endinput
