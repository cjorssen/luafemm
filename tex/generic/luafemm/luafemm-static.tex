% Generic PGF frontiers, winding circuits and static material extensions.
% Picture-level declarations share the boundary queue. Path declarations are
% captured in postactions; they never change the existing femm/boundary key.
\def\luafemm@staticnumber#1#2{%
  \pgfkeys{/pgf/fpu=true}%
  \pgfmathparse{\pgfkeysvalueof{#1}}%
  \pgfmathfloattofixed{\pgfmathresult}\let#2\pgfmathresult
  \pgfkeys{/pgf/fpu=false}}
\def\luafemm@staticoptional#1#2{%
  \edef#2{\pgfkeysvalueof{#1}}%
  \ifx#2\pgfutil@empty\else\luafemm@staticnumber{#1}#2\fi}

\def\luafemm@capturedomain{\begingroup
  \pgfsyssoftpath@getcurrentpath\luafemm@staticpath
  \pgfprocessround{\luafemm@staticpath}{\luafemm@staticpath}%
  \luafemm@staticnumber{/femm/domain/curve tolerance}\luafemm@statictolerance
  \directlua{luafemm.tex.static_path('domain',
    '\luaescapestring{\meaning\luafemm@staticpath}',{
    tolerance=luafemm.tex.number('\luafemm@statictolerance','curve tolerance'),
    fill_rule='\ifpgfsys@eorule even odd\else nonzero\fi'})}%
  \endgroup}
