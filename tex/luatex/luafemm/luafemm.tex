% The only Lua global owned by the package. Numerical modules return tables.
% status.filename identifies the file actually being read, including explicit
% paths. Resolve a bare name through the active format's own resolver only.
\directlua{
  local filename = status.filename
  if not filename:find('/',1,true) then
    if resolvers and resolvers.findfile then
      filename = resolvers.findfile(filename)
    else
      filename = kpse.find_file(filename,'tex')
    end
  end
  local directory = assert(filename:match('^(.+)/[^/]+$'),
    'luafemm: cannot locate the selected TeX installation')
  assert(loadfile(directory..'/lua/luafemm-loader.lua'))().select(directory..'/lua')
  luafemm = require('luafemm')
  luafemm.tex = require('luafemm-tex')
  token.set_macro('luafemm@directory',directory)
}%
% Quoted input names preserve spaces in an installed TEXMF directory.
