% Generic moving-band templates. All numerical expressions are evaluated by PGF,
% then passed as escaped strings; Lua never evaluates user-supplied source text.
\pgfkeys{/femm/problem/sliding band/.initial=,
  /femm/problem/rotor angle/.initial=0,
  /femm/sliding band/.is family,/femm/sliding band,
  name/.initial=,model/.initial=,
  inner radius/.initial=0,outer radius/.initial=0,
  center x/.initial=0,center y/.initial=0,
  segments/.initial=96,layers/.initial=2}
\tikzset{femm/sliding band/.code={%
  \tikzset{execute at end picture={%
    \femmslidingband[#1,model=\luafemm@picturemodel]}}}}
\protected\def\femmslidingband{%
  \pgfutil@ifnextchar[{\luafemm@slidingband}{\luafemm@slidingband[]}}
\def\luafemm@slidingband[#1]{%
  \begingroup
    \pgfqkeys{/femm/sliding band}{name=,model=,inner radius=0,outer radius=0,
      center x=0,center y=0,segments=96,layers=2,#1}%
    \luafemm@staticnumber{/femm/sliding band/inner radius}\luafemm@bandinner
    \luafemm@staticnumber{/femm/sliding band/outer radius}\luafemm@bandouter
    \luafemm@staticnumber{/femm/sliding band/center x}\luafemm@bandx
    \luafemm@staticnumber{/femm/sliding band/center y}\luafemm@bandy
    \luafemm@staticnumber{/femm/sliding band/segments}\luafemm@bandsegments
    \luafemm@staticnumber{/femm/sliding band/layers}\luafemm@bandlayers
    \directlua{luafemm.tex.sliding_band(
      '\luaescapestring{\pgfkeysvalueof{/femm/sliding band/name}}',
      '\luaescapestring{\pgfkeysvalueof{/femm/sliding band/model}}',{
      inner_radius='\luaescapestring{\luafemm@bandinner}',
      outer_radius='\luaescapestring{\luafemm@bandouter}',
      center_x='\luaescapestring{\luafemm@bandx}',center_y='\luaescapestring{\luafemm@bandy}',
      segments='\luaescapestring{\luafemm@bandsegments}',layers='\luaescapestring{\luafemm@bandlayers}'})}%
  \endgroup}
\endinput
