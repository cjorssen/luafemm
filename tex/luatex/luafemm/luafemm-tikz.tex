% Capture the picture frame only after TikZ has installed all its options.
\def\luafemm@starttikz{%
  % Several styles may set femm/problem. Their final values describe one
  % model; repeated begin hooks must not replace it or discard its boundaries.
  \ifluafemm@modelstarted\else
  \pgfgettransform\luafemm@roottransform
  \pgfkeyssetvalue{/femm/region/curve tolerance}{\luafemm@problemvalue{curve tolerance}}%
  \luafemm@responsecurrents
  \luafemm@staticnumber{/femm/problem/rotor angle}\luafemm@rotorangle
  \directlua{
    luafemm.tex.new_problem({
      xmin='\luaescapestring{\luafemm@problemvalue{xmin}}',xmax='\luaescapestring{\luafemm@problemvalue{xmax}}',
      ymin='\luaescapestring{\luafemm@problemvalue{ymin}}',ymax='\luaescapestring{\luafemm@problemvalue{ymax}}',
      applied_field='\luaescapestring{\luafemm@problemvalue{applied field}}',
      applied_phase='\luaescapestring{\luafemm@problemvalue{applied phase}}',
      frequency='\luaescapestring{\luafemm@problemvalue{frequency}}',
      problem_type='\luaescapestring{\luafemm@problemvalue{type}}',
      contour_subdivisions='\luaescapestring{\luafemm@problemvalue{contour subdivisions}}',
      field_model='\luaescapestring{\luafemm@problemvalue{field model}}',
      h='\luaescapestring{\luafemm@problemvalue{mesh size}}',mesher='\luaescapestring{\luafemm@problemvalue{mesher}}',
      mesh_tolerance='\luaescapestring{\luafemm@problemvalue{mesh tolerance}}',
      min_angle='\luaescapestring{\luafemm@problemvalue{minimum angle}}',
      max_area='\luaescapestring{\luafemm@problemvalue{maximum area}}',
      max_steiner='\luaescapestring{\luafemm@problemvalue{max Steiner points}}',
      depth='\luaescapestring{\luafemm@problemvalue{depth}}',
      curve_tolerance='\luaescapestring{\luafemm@problemvalue{curve tolerance}}',
      interpolation='\luaescapestring{\luafemm@problemvalue{interpolation}}',
      tolerance='\luaescapestring{\luafemm@problemvalue{tolerance}}',max_newton='\luaescapestring{\luafemm@problemvalue{max iterations}}'},
      '\luaescapestring{\luafemm@roottransform}',
      {'\the\pgf@xx','\the\pgf@xy','\the\pgf@yx','\the\pgf@yy'},
      {file='\luaescapestring{\pgfkeysvalueof{/femm/cache/file}}',
       mode='\luaescapestring{\pgfkeysvalueof{/femm/cache/mode}}'},
      '\luaescapestring{\pgfkeysvalueof{/femm/import/file}}',
      '\luaescapestring{\luafemm@explicitproblem}',
      '\luaescapestring{\luafemm@problemvalue{name}}',
      {name='\luaescapestring{\luafemm@problemvalue{operating point}}',
       mode='\luaescapestring{\luafemm@problemvalue{permeability}}'},
      {name='\luaescapestring{\luafemm@problemvalue{sliding band}}',
       angle='\luaescapestring{\luafemm@rotorangle}'})}%
  \luafemm@modelstartedtrue
  \luafemm@pendingexcitations
  \let\luafemm@pendingexcitations\pgfutil@empty
  \luafemm@pendingboundaries
  \let\luafemm@pendingboundaries\pgfutil@empty
  \fi
}
\def\luafemm@capturetikz{%
  \begingroup
  \pgfsyssoftpath@getcurrentpath\luafemm@capturedpath
  \pgfprocessround{\luafemm@capturedpath}{\luafemm@capturedpath}%
  \edef\luafemm@library{\luafemm@regionvalue{library}}%
  \pgfqkeys{/tikz/femm/materials}{mu r={},bh={},remanence={},coercivity={},mu x={},mu y={},material angle=0,conductivity={},lamination=inherit,lamination fill={},loss angle={},loss angle x={},loss angle y={},lamination thickness={}}%
  \ifx\luafemm@library\pgfutil@empty
    \pgfqkeys{/tikz/femm/materials}{\luafemm@regionvalue{material}}%
  \fi
  \luafemm@staticoptional{/tikz/femm/materials/mu x}\luafemm@smux
  \luafemm@staticoptional{/tikz/femm/materials/mu y}\luafemm@smuy
  \luafemm@staticoptional{/tikz/femm/materials/material angle}\luafemm@smatangle
  \luafemm@staticoptional{/tikz/femm/materials/conductivity}\luafemm@ssigma
  \luafemm@staticoptional{/tikz/femm/materials/lamination fill}\luafemm@slamfill
  \luafemm@staticoptional{/tikz/femm/materials/loss angle}\luafemm@sloss
  \luafemm@staticoptional{/tikz/femm/materials/loss angle x}\luafemm@slossx
  \luafemm@staticoptional{/tikz/femm/materials/loss angle y}\luafemm@slossy
  \luafemm@staticoptional{/tikz/femm/materials/lamination thickness}\luafemm@sthickness
  \directlua{luafemm.tex.material('\luaescapestring{\luafemm@regionvalue{material}}',
    '\luaescapestring{\luafemm@library}',
    '\luaescapestring{\pgfkeysvalueof{/tikz/femm/materials/mu r}}',
    '\luaescapestring{\pgfkeysvalueof{/tikz/femm/materials/bh}}',
    '\luaescapestring{\pgfkeysvalueof{/tikz/femm/materials/remanence}}',
    '\luaescapestring{\pgfkeysvalueof{/tikz/femm/materials/coercivity}}',{
      loss_angle='\luaescapestring{\luafemm@sloss}',
      loss_angle_x='\luaescapestring{\luafemm@slossx}',
      loss_angle_y='\luaescapestring{\luafemm@slossy}',
      lamination_thickness='\luaescapestring{\luafemm@sthickness}',
      mur_x='\luaescapestring{\luafemm@smux}',
      mur_y='\luaescapestring{\luafemm@smuy}',
      material_angle='\luaescapestring{\luafemm@smatangle}',
      conductivity='\luaescapestring{\luafemm@ssigma}',
      lamination_fill='\luaescapestring{\luafemm@slamfill}',
      lamination='\luaescapestring{\pgfkeysvalueof{/tikz/femm/materials/lamination}}'})}%
  \edef\luafemm@currentinput{\luafemm@regionvalue{current density}}\def\luafemm@frommaterial{material}%
  \ifx\luafemm@currentinput\luafemm@frommaterial
    \edef\luafemm@current{\directlua{tex.sprint(luafemm.current.materials['\luaescapestring{\luafemm@regionvalue{material}}'].current)}}%
  \else
    \pgfkeys{/pgf/fpu=true}%
    \pgfmathparse{\luafemm@currentinput}%
    \pgfmathfloattofixed{\pgfmathresult}\let\luafemm@current\pgfmathresult
    \pgfkeys{/pgf/fpu=false}%
  \fi
  \pgfmathparse{\luafemm@regionvalue{magnetization angle}}\let\luafemm@angle\pgfmathresult
  \luafemm@staticnumber{/femm/region/magnetization center x}\luafemm@scenterx
  \luafemm@staticnumber{/femm/region/magnetization center y}\luafemm@scentery
  \luafemm@staticnumber{/femm/region/magnetization order}\luafemm@sorder
  \luafemm@staticnumber{/femm/region/turns}\luafemm@sturns
  \luafemm@staticnumber{/femm/region/winding fill}\luafemm@sfill
  \luafemm@staticnumber{/femm/region/conductor length}\luafemm@slength
  \directlua{luafemm.tex.capture_path(luafemm.current,
    '\luaescapestring{\meaning\luafemm@capturedpath}',
    '\luaescapestring{\luafemm@regionvalue{material}}',
    luafemm.tex.number('\luaescapestring{\luafemm@current}','current density'),
    luafemm.tex.number('\luaescapestring{\luafemm@regionvalue{curve tolerance}}','curve tolerance'),
    luafemm.tex.number('\luaescapestring{\luafemm@angle}','magnetization angle'),
    luafemm.tex.number('\luaescapestring{\luafemm@regionvalue{mesh size}}','mesh size'),
    '\luaescapestring{\luafemm@componentturns}',
    '\ifpgfsys@eorule even odd\else nonzero\fi',
    '\luaescapestring{\luafemm@regionvalue{ideal domain}}',{current_phase=luafemm.tex.number('\luaescapestring{\luafemm@regionvalue{current phase}}','current phase'),conductor='\luaescapestring{\luafemm@regionvalue{conductor}}',current_constraint='\luaescapestring{\luafemm@regionvalue{current constraint}}',magnetization='\luaescapestring{\luafemm@regionvalue{magnetization}}',circuit='\luaescapestring{\luafemm@regionvalue{circuit}}',winding='\luaescapestring{\luafemm@regionvalue{winding}}',center_x=luafemm.tex.number('\luafemm@scenterx','centerx'),center_y=luafemm.tex.number('\luafemm@scentery','centery'),order=luafemm.tex.number('\luafemm@sorder','order'),turns=luafemm.tex.number('\luafemm@sturns','turns'),fill=luafemm.tex.number('\luafemm@sfill','fill'),length=luafemm.tex.number('\luafemm@slength','length')})}%
  \endgroup
}
