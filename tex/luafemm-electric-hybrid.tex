% An absent strength must cross the adapter as empty, not as numerical zero.
\def\luafemm@sourceoptional#1#2{%
  \edef#2{\pgfkeysvalueof{/femm/electric source/#1}}%
  \ifx#2\pgfutil@empty\else\luafemm@staticnumber{/femm/electric source/#1}#2\fi}
\pgfdeclareshape{femm electric source}{%
  \inheritsavedanchors[from=circle]
  \inheritanchorborder[from=circle]
  \inheritbackgroundpath[from=circle]
  \inheritanchor[from=circle]{center}
  \inheritanchor[from=circle]{text}
  \inheritanchor[from=circle]{base}
  \inheritanchor[from=circle]{mid}
  \inheritanchor[from=circle]{north}
  \inheritanchor[from=circle]{south}
  \inheritanchor[from=circle]{east}
  \inheritanchor[from=circle]{west}
  \inheritanchor[from=circle]{north east}
  \inheritanchor[from=circle]{north west}
  \inheritanchor[from=circle]{south east}
  \inheritanchor[from=circle]{south west}
  \behindbackgroundpath{%
    \ifpgfmatrix\pgferror{luafemm: electric sources require immediate node placement}\else
    \ifpgflatenodepositioning\pgferror{luafemm: electric sources require immediate node placement}\else
      \begingroup
      \luafemm@sourceoptional{charge}\luafemm@q
      \luafemm@sourceoptional{charge per length}\luafemm@lambda
      \luafemm@staticnumber{/femm/electric source/source guard radius}\luafemm@guard
      \luafemm@staticnumber{/femm/electric source/exclusion radius}\luafemm@radius
      \pgfgettransform\luafemm@electrictransform
      \centerpoint
      \directlua{require('luafemm-electric-tex').source('\luafemm@electricid',
        '\luaescapestring{\luafemm@electrictransform}','\the\pgf@x','\the\pgf@y',{
        kind='\luaescapestring{\pgfkeysvalueof{/femm/electric source/kind}}',
        charge='\luaescapestring{\luafemm@q}',charge_per_length='\luaescapestring{\luafemm@lambda}',
        guard_radius='\luaescapestring{\luafemm@guard}',radius='\luaescapestring{\luafemm@radius}'})}%
      \endgroup
    \fi\fi}}
