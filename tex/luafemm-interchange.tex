\pgfkeys{/femm/copy answer/.is family,/femm/copy answer,
  input/.initial=,output/.initial=,rings/.initial=preserve,rings/.is choice,
  rings/preserve/.code={\pgfkeyssetvalue{/femm/copy answer/rings}{preserve}},
  rings/omit/.code={\pgfkeyssetvalue{/femm/copy answer/rings}{omit}}}
\protected\def\femmcopyanswer{\pgfutil@ifnextchar[{\luafemm@copyanswer}{\luafemm@copyanswer[]}}
\def\luafemm@copyanswer[#1]{\begingroup
  \pgfqkeys{/femm/copy answer}{input=,output=,rings=preserve,#1}%
  \directlua{luafemm.tex.copy_answer(
    '\luaescapestring{\pgfkeysvalueof{/femm/copy answer/input}}',
    '\luaescapestring{\pgfkeysvalueof{/femm/copy answer/output}}',
    '\luaescapestring{\pgfkeysvalueof{/femm/copy answer/rings}}')}%
  \endgroup}
\tikzset{femm/copy answer/.code={\femmcopyanswer[#1]}}
