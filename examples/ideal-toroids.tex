\begin{document}
\centering

  \node[below] at (M.south) {Closed toroid};
\end{tikzpicture}
\quad
\begin{tikzpicture}[ideal toroid,scale=.7]
  \node[femm/toroid={ampere turns=1200,gap angle=8}] (M) {};

  \pic[femm/mean={component=M}] {femm mean path};
  \node[below] at (M.south) {Confined gap};
\end{tikzpicture}

\begin{tikzpicture}[ideal toroid,scale=.7]
  \node[femm/tapered toroid={ampere turns=1200,

  \pic[femm/lines=9] {femm field};
  \pic[femm/mean={component=M,profile=taper-coil}] {femm mean path};
  \node[below] at (M.south) {Tapered poles and winding};

\quad
\begin{tikzpicture}[ideal toroid,scale=.7]
